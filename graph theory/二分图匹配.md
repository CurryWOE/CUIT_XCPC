# 无权二分图最大匹配
## Kuhn算法
[测试网站1](https://judge.yosupo.jp/problem/bipartitematching)的[提交](https://judge.yosupo.jp/submission/396198)971ms

[测试网站2](https://www.luogu.com.cn/problem/P3386)的[提交](https://www.luogu.com.cn/record/294762981)46ms是速度第一
```cpp
namespace Kuhn//0-index
{
    vector<int> e[N];
    int match_left[N], match_right[N], seen[N], timer;
    bool dfs(int u)
    {
        for (auto v : e[u])
        {
            if (match_right[v] != -1)
                continue;
            match_left[u] = v;
            match_right[v] = u;
            return true;
        }
        for (auto v : e[u])
        {
            if (seen[v] == timer)
                continue;
            seen[v] = timer;
            if (!dfs(match_right[v]))
                continue;
            match_left[u] = v;
            match_right[v] = u;
            return true;
        }
        return false;
    }
    int run(int leftPointNumber, int rightPointNumber, vector<array<int, 2>> edge)
    {
        auto mi = min(leftPointNumber, rightPointNumber), ma = max(leftPointNumber, rightPointNumber);
        fill(match_left, match_left + mi, -1);
        fill(match_right, match_right + ma, -1);
        fill(seen, seen + ma, 0);
        timer = 1;
        for (int i = 0; i < mi; ++i)
            e[i].clear();
        for (auto op = (leftPointNumber > rightPointNumber); auto ed : edge)
            e[ed[op]].push_back(ed[!op]);
        for (int i = 0; i < mi; ++i)
            ranges::sort(e[i]);
        vector<int> order(mi);
        ranges::iota(order, 0);
        ranges::sort(order, [&](int u, int v){ return e[u].size() < e[v].size(); });
        auto maximum_matching = 0;
        for (bool any_augmented = true; any_augmented; ++timer)
        {
            any_augmented = false;
            for (auto u : order)
                if (match_left[u] == -1 && dfs(u))
                {
                    ++maximum_matching;
                    any_augmented = true;
                }
        }
        return maximum_matching;
        /*谁和谁匹配的方案：
        for (int i = 0; i < mi; ++i)
            if (match_left[i] != -1)
                cout<<(mi==leftPointNumber?i:match_left[i])<<' '<<(mi==leftPointNumber?match_left[i]:i)<<'\n';
        */
    }
}
```
