# 无权二分图最大匹配
## Kuhn算法
$O(V\times E)$

[测试网站1](https://judge.yosupo.jp/problem/bipartitematching)的[提交](https://judge.yosupo.jp/submission/396198)971ms

[测试网站2](https://www.luogu.com.cn/problem/P3386)的[提交](https://www.luogu.com.cn/record/294762981)46ms是速度第一
```cpp
namespace Kuhn//0-index
{
    vector<int> e[N];
    int match_left[N], match_right[N], seen[N], timer;
    bool dfs(int u)
    {
        for (auto v : e[u])
        {
            if (match_right[v] != -1)
                continue;
            match_left[u] = v;
            match_right[v] = u;
            return true;
        }
        for (auto v : e[u])
        {
            if (seen[v] == timer)
                continue;
            seen[v] = timer;
            if (!dfs(match_right[v]))
                continue;
            match_left[u] = v;
            match_right[v] = u;
            return true;
        }
        return false;
    }
    int run(int leftPointNumber, int rightPointNumber, vector<array<int, 2>> edge)
    {
        auto mi = min(leftPointNumber, rightPointNumber), ma = max(leftPointNumber, rightPointNumber);
        fill(match_left, match_left + mi, -1);
        fill(match_right, match_right + ma, -1);
        fill(seen, seen + ma, 0);
        timer = 1;
        for (int i = 0; i < mi; ++i)
            e[i].clear();
        for (auto op = (leftPointNumber > rightPointNumber); auto ed : edge)
            e[ed[op]].push_back(ed[!op]);
        for (int i = 0; i < mi; ++i)
            ranges::sort(e[i]);
        vector<int> order(mi);
        ranges::iota(order, 0);
        ranges::sort(order, [&](int u, int v){ return e[u].size() < e[v].size(); });
        auto maximum_matching = 0;
        for (bool any_augmented = true; any_augmented; ++timer)
        {
            any_augmented = false;
            for (auto u : order)
                if (match_left[u] == -1 && dfs(u))
                {
                    ++maximum_matching;
                    any_augmented = true;
                }
        }
        return maximum_matching;
        /*谁和谁匹配的方案：
        for (int i = 0; i < mi; ++i)
            if (match_left[i] != -1)
                cout<<(mi==leftPointNumber?i:match_left[i])<<' '<<(mi==leftPointNumber?match_left[i]:i)<<'\n';
        */
    }
}
```
## Hopcroft–Karp算法
$O(\sqrt V E)$

[测试网站1](https://judge.yosupo.jp/problem/bipartitematching)的[提交](https://judge.yosupo.jp/submission/396318)70ms是速度第一
```cpp
namespace HopcroftKarp
{
    uint32_t match[N], dist[N], leftVertices[N];
    constexpr uint32_t UNSET=-1;
    uint32_t run(uint32_t leftPointCount, uint32_t totalPointCount)
    {
        //如果左右点都是1开始编号，右部点需要从leftPointCount+1开始重编号
        fill(match, match + totalPointCount, UNSET);
        ranges::iota(leftVertices, leftVertices + leftPointCount, 0);
        queue<uint32_t> augmentQueue, bfs;
        for (auto i = leftPointCount; i < totalPointCount; ++i)
            augmentQueue.push(i);
        for (uint32_t i = 0;;)
        {
            if (i == 0)
            {
                fill(dist, dist + totalPointCount, totalPointCount);
                for (auto u : leftVertices)
                    if (match[u] == UNSET)
                    {
                        dist[u] = 0;
                        bfs.push(u);
                    }
                while (!bfs.empty())
                {
                    uint32_t u = bfs.front();
                    bfs.pop();
                    for (auto v : e[u])
                    {
                        if (dist[v] <= dist[u] + 1)
                            continue;
                        dist[v] = dist[u] + 1;
                        if (auto w = match[v]; w != UNSET)
                        {
                            dist[w] = dist[v] + 1;
                            bfs.push(w);
                        }
                    }
                }
            }
            if (augmentQueue.empty())
                break;
            uint32_t u = augmentQueue.front(), bestLeft = UNSET, minDist = numeric_limits<uint32_t>::max();
            augmentQueue.pop();
            for (auto v : e[u])
                if (dist[v] < minDist)
                {
                    minDist = dist[v];
                    bestLeft = u;
                }
            if (bestLeft != UNSET && dist[bestLeft] < totalPointCount)
            {
                dist[u] = dist[bestLeft] + 1;
                if (auto w = match[bestLeft]; w != UNSET)
                {
                    match[w] = UNSET;
                    augmentQueue.push(w);
                }
                match[u] = bestLeft;
                match[bestLeft] = u;
                dist[bestLeft] += 2;
            }
            if (++i == totalPointCount)
                i = 0;
        }
        return (totalPointCount - ranges::count(match, match + totalPointCount, UNSET)) / 2;
        /*谁和谁匹配的方案：
        for (int i = 0; i < leftPointCount; ++i)
            if (match[i] != UNSET)
                cout << i << ' ' << match[i] - leftPointCount << '\n';
        */
    }
}
```
